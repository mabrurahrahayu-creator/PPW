import requests
from bs4 import BeautifulSoup
import pandas as pd
from urllib.parse import urlparse
import time

def scrape_kompas_article(url):
    # Header standard untuk meniru browser
    headers = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36"}

    # Inisialisasi data_artikel
    data_artikel = {
        "Judul": "N/A",
        "Isi (100 kata)": "N/A",
        "Kategori": "N/A",
        "URL": url,
        "Status": "Gagal"
    }

    try:
        # Lakukan request ke URL dengan timeout
        r = requests.get(url, headers=headers, timeout=10)
        r.raise_for_status() # Tangkap error HTTP (4xx atau 5xx)

    except requests.exceptions.RequestException as e:
        data_artikel["Status"] = f"Error Request: {e}"
        print(f"-> Gagal mengambil URL {url[:50]}...: {e}")
        return data_artikel

    soup = BeautifulSoup(r.content, "html.parser")

    # --- Ekstraksi Data ---

    # Judul (Selector yang umum digunakan Kompas untuk judul)
    judul = soup.select_one("h1.read__title")
    data_artikel["Judul"] = judul.text.strip() if judul else "Tidak ditemukan judul"

    # Isi (100 kata)
    isi_elems = soup.select("div.read__content p")
    isi_text = " ".join([p.get_text(strip=True) for p in isi_elems])
    words = isi_text.split()
    isi_100 = " ".join(words[:100]) + ("..." if len(words) > 100 else "")
    data_artikel["Isi (100 kata)"] = isi_100

    # Kategori: Ditemukan dari 3 metode (meta tag, breadcrumb, atau subdomain)
    kategori_text = "Tidak ditemukan kategori"

    # 1. Coba dari meta tag
    kategori_meta = soup.find("meta", {"property": "article:section"})
    if kategori_meta and kategori_meta.get("content"):
        kategori_text = kategori_meta["content"].strip()

    # 2. Coba dari breadcrumb jika meta tag gagal
    if kategori_text == "Tidak ditemukan kategori":
        breadcrumb = soup.select("div.breadcrumb__link a")
        if breadcrumb and len(breadcrumb) > 1:
            kategori_text = breadcrumb[1].get_text(strip=True)

    # 3. Coba dari subdomain jika breadcrumb gagal
    if kategori_text == "Tidak ditemukan kategori":
        parsed = urlparse(url)
        subdomain = parsed.netloc.split(".")[0]
        if subdomain and subdomain != "www" and subdomain != "kompas":
            kategori_text = subdomain.capitalize()

    data_artikel["Kategori"] = kategori_text
    data_artikel["Status"] = "Sukses"

    print(f"-> Sukses: {data_artikel['Judul'][:50]}...")
    return data_artikel
